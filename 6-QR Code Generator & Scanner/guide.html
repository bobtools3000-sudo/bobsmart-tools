// وظيفة التوليد function generateQR() { var text = document.getElementById('textInput').value; var qrcodeDiv = document.getElementById("qrcode"); qrcodeDiv.innerHTML = ""; // تنظيف المكان if(text.trim() !== "") { new QRCode(qrcodeDiv, { text: text, width: 200, height: 200, correctLevel: QRCode.CorrectLevel.H }); } else { alert("يرجى إدخال نص!"); } } // وظيفة المسح function startScan() { const scanner = new Html5QrcodeScanner("reader", { fps: 10, qrbox: 250 }); scanner.render((text) => { alert("تمت القراءة بنجاح: " + text); scanner.clear(); // إغلاق الكاميرا بعد القراءة }); }
